# Platform prologue. Prepended to every repo's skipfile and NOT overridable:
# CodeChecker skipfiles are first-match-wins, so these rules are decided before
# anything the repo says. Scope is repo knowledge, but "you may not exclude the
# file that has the findings" is policy.
-*/build*/*
-*/cmake-build*/*
-*/third_party/*
-*/thirdparty/*
-*/3rdparty/*
-*/deps/*
-*/external/*
# Installed and system headers: a repo's "+*/include/*" would otherwise pull in
# /usr/include and bury its own findings in library noise.
-/usr/*
-/opt/*
