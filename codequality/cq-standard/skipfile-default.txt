# whose sources live elsewhere should ship one rather than rely on this guess.
#

# checked out - and "-*" closes the allow-list, so anything not named below is
# out, system headers included.

-/src/tests/*
-/src/test/*
-/src/*/tests/*

+/src/src/*
+/src/include/*
+/src/daemon/*
+/src/lib/*
